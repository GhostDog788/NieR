cmake_minimum_required(VERSION 3.20)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(NIER_BUILD_PUBLISHER "Build stock-Clang publisher integration" ON)
option(NIER_BUILD_TESTS "Build NieR tests" ON)
if(DEFINED ENV{NIER_SDK_ROOT})
  set(NIER_INITIAL_SDK "$ENV{NIER_SDK_ROOT}")
else()
  set(NIER_INITIAL_SDK "${CMAKE_SOURCE_DIR}/.sdk")
endif()
set(NIER_SDK_ROOT "${NIER_INITIAL_SDK}" CACHE PATH "Verified local SDK")
if(NOT DEFINED CMAKE_C_COMPILER AND NOT DEFINED ENV{CC})
  set(CMAKE_C_COMPILER "${NIER_SDK_ROOT}/host/usr/lib/llvm-18/bin/clang" CACHE FILEPATH "C compiler")
endif()
if(NOT DEFINED CMAKE_CXX_COMPILER AND NOT DEFINED ENV{CXX})
  set(CMAKE_CXX_COMPILER "${NIER_SDK_ROOT}/host/usr/lib/llvm-18/bin/clang++" CACHE FILEPATH "C++ compiler")
endif()
project(Nier VERSION 0.0.1 LANGUAGES C CXX)
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  add_link_options("--ld-path=${NIER_SDK_ROOT}/host/usr/lib/llvm-18/bin/ld.lld")
endif()
file(SHA256 "${CMAKE_SOURCE_DIR}/sdk/packages.lock" NIER_SDK_LOCK_SHA256)
list(PREPEND CMAKE_PREFIX_PATH "${NIER_SDK_ROOT}/host/usr/lib/llvm-18")
find_package(LLVM 18.1.3 EXACT REQUIRED CONFIG)
find_package(MLIR REQUIRED CONFIG)
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
set(CMAKE_BUILD_RPATH "${NIER_SDK_ROOT}/host/usr/lib/llvm-18/lib;${NIER_SDK_ROOT}/host/usr/lib/x86_64-linux-gnu")
find_path(NIER_ARCHIVE_INCLUDE archive.h HINTS "${NIER_SDK_ROOT}/host/usr/include" REQUIRED)
find_library(NIER_ARCHIVE_LIBRARY archive HINTS "${NIER_SDK_ROOT}/host/usr/lib/x86_64-linux-gnu" REQUIRED)

add_library(nier_support STATIC src/support/Support.cpp)
target_include_directories(nier_support PUBLIC include)
target_link_libraries(nier_support PUBLIC LLVM)
target_compile_definitions(nier_support PRIVATE NIER_SDK_LOCK_SHA256="${NIER_SDK_LOCK_SHA256}")
add_library(nier_artifact STATIC src/artifact/Artifact.cpp)
target_include_directories(nier_artifact PRIVATE ${NIER_ARCHIVE_INCLUDE})
target_link_libraries(nier_artifact PUBLIC nier_support PRIVATE ${NIER_ARCHIVE_LIBRARY})
add_library(nier_ir STATIC src/ir/Compiler.cpp src/ir/Dialect.cpp src/ir/AggregateABI.cpp src/ir/AggregateLayoutABI.cpp src/ir/OverlapLayout.cpp src/ir/CompilationUnits.cpp src/ir/ConditionalSpecialization.cpp src/ir/NativeABIBridge.cpp)
target_include_directories(nier_ir PUBLIC include)
target_link_libraries(nier_ir PUBLIC MLIRIR MLIRParser MLIRBytecodeReader MLIRBytecodeWriter LLVM)
add_executable(nierc src/consumer/Main.cpp)
target_link_libraries(nierc PRIVATE nier_ir nier_artifact)
target_compile_definitions(nierc PRIVATE NIER_DEFAULT_SDK="${NIER_SDK_ROOT}")
# Our installed compiler uses a transitive private RPATH. Stock LLVM tools
# remain unmodified and receive their matching libraries in child-only env.
target_link_options(nierc PRIVATE "LINKER:--disable-new-dtags")
set_target_properties(nierc PROPERTIES INSTALL_RPATH "$ORIGIN/../sdk/host/usr/lib/llvm-18/lib")
install(TARGETS nierc RUNTIME DESTINATION bin COMPONENT Consumer)
install(DIRECTORY include/nier DESTINATION include)

if(NIER_BUILD_PUBLISHER)
  find_package(Clang REQUIRED CONFIG)
  add_library(nier_llvm_producer STATIC src/ir/Producer.cpp src/ir/Varargs.cpp src/ir/ByteSwap.cpp src/ir/OverlapEvidence.cpp src/ir/Partitions.cpp src/ir/ConditionalCFG.cpp src/ir/AggregateNormalize.cpp)
  target_link_libraries(nier_llvm_producer PUBLIC nier_ir)
  add_library(nier_capture MODULE src/capture/Snapshot.cpp)
  target_link_libraries(nier_capture PRIVATE LLVM)
  set_target_properties(nier_capture PROPERTIES PREFIX "" OUTPUT_NAME "nier-capture")
  add_library(nier_clang MODULE src/publisher/ClangPlugin.cpp)
  target_include_directories(nier_clang SYSTEM PRIVATE ${CLANG_INCLUDE_DIRS})
  target_link_libraries(nier_clang PRIVATE clang-cpp nier_llvm_producer nier_artifact)
  target_compile_definitions(nier_clang PRIVATE NIER_DEFAULT_SDK="${NIER_SDK_ROOT}" NIER_CAPTURE_PLUGIN="$<TARGET_FILE:nier_capture>")
  set_target_properties(nier_clang PROPERTIES OUTPUT_NAME "nier-clang")
  add_dependencies(nier_clang nier_capture)
  add_executable(nier-ld src/publisher/LinkMain.cpp)
  target_link_libraries(nier-ld PRIVATE nier_artifact nier_ir)
  add_library(nier_build STATIC src/cli/Build.cpp)
  target_link_libraries(nier_build PUBLIC nier_support)
  target_include_directories(nier_build PUBLIC src/cli)
  target_compile_definitions(nier_build PRIVATE NIER_CLANG_PLUGIN="$<TARGET_FILE:nier_clang>" NIER_CAPTURE_PLUGIN="$<TARGET_FILE:nier_capture>" NIER_DEFAULT_CONFIG="${CMAKE_BINARY_DIR}/nier.cfg")
  add_executable(nier-build src/publisher/BuildMain.cpp)
  target_link_libraries(nier-build PRIVATE nier_build nier_artifact)
  target_compile_definitions(nier-build PRIVATE NIER_DEFAULT_SDK="${NIER_SDK_ROOT}" NIER_DEFAULT_CONFIG="${CMAKE_BINARY_DIR}/nier.cfg")
  add_executable(nier-native-ld src/publisher/NativeLinkMain.cpp)
  target_link_libraries(nier-native-ld PRIVATE nier_build)
  file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/nier.cfg" CONTENT
"--target=x86_64-unknown-linux-gnu
--sysroot=\"${NIER_SDK_ROOT}/sysroots/x86_64-linux-gnu\"
-resource-dir=\"${NIER_SDK_ROOT}/host/usr/lib/llvm-18/lib/clang/18\"
-fPIC
-nostdlib
--ld-path=\"$<TARGET_FILE:nier-ld>\"
-Xclang -load -Xclang \"$<TARGET_FILE:nier_clang>\"
-Xclang -plugin -Xclang nier
-Xclang -plugin-arg-nier -Xclang \"sdk=${NIER_SDK_ROOT}\"
")
  install(TARGETS nier_clang nier_capture nier-ld nier-build nier-native-ld LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
endif()
