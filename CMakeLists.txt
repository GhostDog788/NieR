cmake_minimum_required(VERSION 3.20)
project(OnTargetAOT VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(DEFINED ENV{AOT_SDK_ROOT})
  set(AOT_INITIAL_SDK "$ENV{AOT_SDK_ROOT}")
else()
  set(AOT_INITIAL_SDK "${CMAKE_SOURCE_DIR}/.sdk")
endif()
set(AOT_SDK_ROOT "${AOT_INITIAL_SDK}" CACHE PATH "Verified local SDK")
file(SHA256 "${CMAKE_SOURCE_DIR}/sdk/packages.lock" AOT_SDK_LOCK_SHA256)
list(PREPEND CMAKE_PREFIX_PATH "${AOT_SDK_ROOT}/host/usr/lib/llvm-18")
find_package(LLVM 18.1.3 EXACT REQUIRED CONFIG)
find_package(MLIR REQUIRED CONFIG)
if(NOT LLVM_PACKAGE_VERSION VERSION_EQUAL "18.1.3")
  message(FATAL_ERROR "This experimental compiler contract requires LLVM 18.1.3")
endif()
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})

add_library(aot_capture MODULE src/capture/Snapshot.cpp)
target_link_libraries(aot_capture PRIVATE LLVM)
set_target_properties(aot_capture PROPERTIES PREFIX "" OUTPUT_NAME "aot-capture")

add_library(aot_ir STATIC src/ir/Compiler.cpp src/ir/Dialect.cpp)
target_include_directories(aot_ir PUBLIC include)
target_link_libraries(aot_ir PUBLIC MLIRIR MLIRParser MLIRBytecodeReader MLIRBytecodeWriter LLVM)

find_path(AOT_ARCHIVE_INCLUDE archive.h HINTS "${AOT_SDK_ROOT}/host/usr/include" REQUIRED)
find_library(AOT_ARCHIVE_LIBRARY archive HINTS "${AOT_SDK_ROOT}/host/usr/lib/x86_64-linux-gnu" REQUIRED)
add_executable(aot src/cli/Main.cpp src/cli/Support.cpp src/cli/Package.cpp src/cli/Build.cpp)
target_include_directories(aot PRIVATE src/cli ${AOT_ARCHIVE_INCLUDE})
target_link_libraries(aot PRIVATE aot_ir ${AOT_ARCHIVE_LIBRARY})
target_compile_definitions(aot PRIVATE AOT_DEFAULT_SDK="${AOT_SDK_ROOT}" AOT_CAPTURE_PLUGIN="$<TARGET_FILE:aot_capture>" AOT_SDK_LOCK_SHA256="${AOT_SDK_LOCK_SHA256}")
add_dependencies(aot aot_capture)

enable_testing()
add_executable(aot_ir_tests tests/ir.cpp)
target_link_libraries(aot_ir_tests PRIVATE aot_ir)
add_test(NAME ir_validation COMMAND aot_ir_tests)
add_executable(package_tests tests/package.cpp src/cli/Package.cpp src/cli/Support.cpp)
target_include_directories(package_tests PRIVATE src/cli ${AOT_ARCHIVE_INCLUDE})
target_link_libraries(package_tests PRIVATE LLVM ${AOT_ARCHIVE_LIBRARY})
target_compile_definitions(package_tests PRIVATE AOT_SDK_LOCK_SHA256="${AOT_SDK_LOCK_SHA256}")
add_test(NAME package_validation COMMAND package_tests)
add_test(NAME capture_hook COMMAND bash ${CMAKE_SOURCE_DIR}/tests/capture.sh $<TARGET_FILE:aot_capture> ${AOT_SDK_ROOT})
add_test(NAME hello_pipeline COMMAND bash ${CMAKE_SOURCE_DIR}/tests/hello.sh $<TARGET_FILE:aot> ${AOT_SDK_ROOT})
add_test(NAME scalar_pipeline COMMAND bash ${CMAKE_SOURCE_DIR}/tests/scalars.sh $<TARGET_FILE:aot> ${AOT_SDK_ROOT})
add_test(NAME build_integration COMMAND bash ${CMAKE_SOURCE_DIR}/tests/build-integration.sh $<TARGET_FILE:aot> ${AOT_SDK_ROOT})
find_program(AOT_STRACE strace)
if(AOT_STRACE)
  add_test(NAME device_boundary COMMAND bash ${CMAKE_SOURCE_DIR}/tests/device-boundary.sh $<TARGET_FILE:aot> ${AOT_SDK_ROOT})
endif()
set_tests_properties(ir_validation package_validation capture_hook hello_pipeline scalar_pipeline build_integration PROPERTIES TIMEOUT 120)
if(AOT_STRACE)
  set_tests_properties(device_boundary PROPERTIES TIMEOUT 120)
endif()
