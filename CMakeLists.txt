function(sela_add_ir_library name)
  add_library(${name} STATIC ${SELA_COMMON_IR_SOURCES})
  target_include_directories(${name} PUBLIC include)
  target_link_libraries(${name} PUBLIC MLIRIR MLIRParser MLIRBytecodeReader MLIRBytecodeWriter ${SELA_LLVM_LIBRARIES})
  set(has_x86_64 0)
  set(has_i686 0)
  foreach(native_target IN LISTS ARGN)
    target_sources(${name} PRIVATE $<TARGET_OBJECTS:sela_native_${native_target}>)
    set(has_${native_target} 1)
  endforeach()
  target_compile_definitions(${name} PRIVATE SELA_HAS_X86_64=${has_x86_64} SELA_HAS_I686=${has_i686})
endfunction()
