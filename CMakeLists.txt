function(sela_add_ir_library name)
  add_library(${name} STATIC ${SELA_COMMON_IR_SOURCES})
  target_include_directories(${name} PUBLIC include)
  target_link_libraries(${name} PUBLIC sela_support MLIRIR MLIRParser MLIRBytecodeReader MLIRBytecodeWriter ${SELA_LLVM_LIBRARIES})
  set(native_registry "")
  foreach(native_target IN LISTS ARGN)
    target_sources(${name} PRIVATE $<TARGET_OBJECTS:sela_native_${native_target}>)
    string(APPEND native_registry "SELA_NATIVE_BACKEND(${native_target})\n")
  endforeach()
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/targets/${name}/sela/LinkedNativeTargets.inc" CONTENT "${native_registry}")
  target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/targets/${name}")
endfunction()
